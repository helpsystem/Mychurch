7.1 HIDRIVE STORAGE SERVICE (backend/services/hidriveStorage.js)
-----------------------------------------------------------------
const axios = require('axios');

class HiDriveStorage {
  constructor() {
    this.baseUrl = 'https://webdav.hidrive.ionos.com/users/adminchurch/mychurch';
    this.auth = {
      username: process.env.HIDRIVE_USER,
      password: process.env.HIDRIVE_PASSWORD
    };
  }

  // Stream file from HiDrive to client
  async streamToResponse(filePath, res) {
    const url = `${this.baseUrl}/${filePath}`;
    
    const response = await axios({
      method: 'get',
      url: url,
      responseType: 'stream',
      auth: this.auth
    });
    
    res.setHeader('Content-Type', response.headers['content-type']);
    res.setHeader('Accept-Ranges', 'bytes');
    res.setHeader('Content-Length', response.headers['content-length']);
    
    response.data.pipe(res);
  }

  // Upload file to HiDrive
  async uploadFile(localPath, remotePath) {
    const url = `${this.baseUrl}/${remotePath}`;
    const fileStream = fs.createReadStream(localPath);
    
    await axios({
      method: 'put',
      url: url,
      data: fileStream,
      auth: this.auth,
      headers: {
        'Content-Type': 'application/octet-stream'
      }
    });
  }
}

module.exports = new HiDriveStorage();


7.2 URL CONVERSION MIDDLEWARE (backend/middleware/urlConverter.js)
-------------------------------------------------------------------
// تبدیل URL مستقیم به proxy URL
function convertToProxyURL(url) {
  if (!url) return url;
  
  // Already a proxy URL
  if (url.startsWith('/api/hidrive/stream/')) {
    return url;
  }
  
  // Extract path from WebDAV URL
  const webdavMatch = url.match(/users\/adminchurch\/mychurch\/(.+)/);
  if (webdavMatch) {
    return `/api/hidrive/stream/${webdavMatch[1]}`;
  }
  
  // Direct WebDAV URL
  const HIDRIVE_PUBLIC_URL = 'https://webdav.hidrive.ionos.com/users/adminchurch/mychurch';
  if (url.startsWith('/worship/') || url.startsWith('/bible/')) {
    return `${HIDRIVE_PUBLIC_URL}${url}`;
  }
  
  return url;
}

module.exports = { convertToProxyURL };

10. NGINX Configuration
================================================================================

File: /etc/nginx/sites-available/samanabyar.online

server {
    listen 80;
    listen [::]:80;
    server_name samanabyar.online www.samanabyar.online;
    
    # Redirect HTTP to HTTPS
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name samanabyar.online www.samanabyar.online;
    
    # SSL Certificates
    ssl_certificate /etc/letsencrypt/live/samanabyar.online/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/samanabyar.online/privkey.pem;
    
    # SSL Configuration
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;
    ssl_prefer_server_ciphers on;
    
    # Root directory
    root /var/www/html;
    index index.html;
    
    # Frontend (Static files)
    location / {
        try_files $uri $uri/ /index.html;
        add_header Cache-Control "public, max-age=3600";
    }
    
    # Backend API (Proxy)
    location /api/ {
        proxy_pass http://127.0.0.1:3001;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
        
        # Timeout for streaming
        proxy_read_timeout 300s;
        proxy_send_timeout 300s;
    }
    
    # Static assets caching
    location ~* \.(jpg|jpeg|png|gif|ico|css|js|svg|woff|woff2|ttf|eot)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
    }
    
    # Security headers
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "1; mode=block" always;
    
    # Max upload size
    client_max_body_size 50M;
}

export const LanguageProvider: React.FC<{ children: ReactNode }> = ({ children }) => {
  const [lang, setLang] = useState<Language>('fa');

  const t = (key: string): string => {
    return getTranslation(key, lang);
  };

  return (
    <LanguageContext.Provider value={{ lang, setLang, t }}>
      {children}
    </LanguageContext.Provider>
  );
};
